Base URL: http://localhost:3000

# 1. GET / - 200
curl -i http://localhost:3000/

# 2. GET /api/v1/health - 200 OK (503 jika DB mati, pool.Ping main.go:61)
curl -i http://localhost:3000/api/v1/health

# 3. POST /api/v1/students/ - 201 Created
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: application/json" -d "{\"nim\":\"22010101\",\"name\":\"Budi\",\"grade\":85.5,\"is_active\":true}"

# 4. POST /api/v1/students/ - 409 Conflict (NIM ganda)
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: application/json" -d "{\"nim\":\"22010101\",\"name\":\"Budi2\",\"grade\":90,\"is_active\":true}"

# 5. POST /api/v1/students/ - 422 nim kosong
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: application/json" -d "{\"nim\":\"\",\"name\":\"Budi\",\"grade\":85.5,\"is_active\":true}"

# 6. POST /api/v1/students/ - 422 name kosong
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: application/json" -d "{\"nim\":\"22010102\",\"name\":\"\",\"grade\":85.5,\"is_active\":true}"

# 7. POST /api/v1/students/ - 422 grade 0-100
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: application/json" -d "{\"nim\":\"22010102\",\"name\":\"Andi\",\"grade\":150,\"is_active\":true}"

# 8. POST /api/v1/students/ - 415 tanpa JSON header
curl -i -X POST http://localhost:3000/api/v1/students/ -H "Content-Type: text/plain" -d "{\"nim\":\"22010103\",\"name\":\"Andi\",\"grade\":85.5,\"is_active\":true}"

# 9. POST /api/v1/students/ - 400 JSON tidak valid
